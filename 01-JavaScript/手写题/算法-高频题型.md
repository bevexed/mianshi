## 一、双指针

### 对撞指针（有序数组）

```js
function twoSumSorted(nums, target) {
	let l = 0,
		r = nums.length - 1;
	while (l < r) {
		const sum = nums[l] + nums[r];
		if (sum === target) return [l, r];
		sum < target ? l++ : r--;
	}
	return [];
}
```

### 快慢指针（原地移除）

```js
function removeDuplicates(nums) {
	// 26. 删除有序数组重复项
	if (!nums.length) return 0;
	let slow = 0;
	for (let fast = 1; fast < nums.length; fast++) {
		if (nums[fast] !== nums[slow]) nums[++slow] = nums[fast];
	}
	return slow + 1;
}
```

**代表题**：11 盛最多水的容器、15 三数之和、283 移动零

## 二、滑动窗口

**识别信号**：「**连续**子数组/子串」+「最长/最短/满足条件」

```js
function lengthOfLongestSubstring(s) {
	// 3. 无重复字符最长子串（超高频）
	const map = new Map();
	let left = 0,
		max = 0;
	for (let right = 0; right < s.length; right++) {
		const c = s[right];
		// 关键：left 只能右移，不能因为旧记录而后退
		if (map.has(c) && map.get(c) >= left) left = map.get(c) + 1;
		map.set(c, right);
		max = Math.max(max, right - left + 1);
	}
	return max;
}
```

**通用模板**：

```js
let left = 0;
for (let right = 0; right < n; right++) {
	// 1. right 加入窗口
	while (窗口不满足条件) {
		// 2. 收缩左边界
		left++;
	}
	// 3. 更新答案
}
```

**代表题**：3 无重复最长子串、209 长度最小子数组、438 找异位词

## 三、前缀和

前缀和把连续区间求和转成两个前缀的差。对于“和为 k 的连续子数组个数”，遍历到当前位置的前缀和为 `sum` 时，只需统计此前出现过多少个 `sum - k`。

```js
function subarraySum(nums, k) {
	const count = new Map([[0, 1]]);
	let sum = 0;
	let result = 0;

	for (const num of nums) {
		sum += num;
		result += count.get(sum - k) ?? 0;
		count.set(sum, (count.get(sum) ?? 0) + 1);
	}

	return result;
}

subarraySum([1, 1, 1], 2); // 2
```

**复杂度**：时间 O(n)，空间 O(n)。初始化 `0 → 1` 用来统计从数组开头就满足条件的区间。数组含负数时滑动窗口通常不再成立，前缀和 + 哈希仍适用。

## 四、哈希表

**识别信号**：「出现次数」「是否存在」「配对」——空间换时间

```js
function twoSum(nums, target) {
	// 1. 两数之和（最高频）
	const map = new Map();
	for (let i = 0; i < nums.length; i++) {
		const need = target - nums[i];
		if (map.has(need)) return [map.get(need), i];
		map.set(nums[i], i);
	}
}

function groupAnagrams(strs) {
	// 49. 字母异位词分组
	const map = new Map();
	for (const s of strs) {
		const key = [...s].sort().join('');
		map.set(key, [...(map.get(key) || []), s]);
	}
	return [...map.values()];
}
```

**代表题**：1 两数之和、217 存在重复、349 数组交集、242 有效字母异位词

## 五、栈

**识别信号**：「括号匹配」「相邻消除」「路径简化」「单调性」

```js
function isValid(s) {
	// 20. 有效的括号
	const pairs = { ')': '(', ']': '[', '}': '{' };
	const opens = new Set(Object.values(pairs));
	const stack = [];
	for (const c of s) {
		if (opens.has(c)) stack.push(c);
		else if (!(c in pairs) || stack.pop() !== pairs[c]) return false;
	}
	return stack.length === 0;
}

function removeAdjacent(s) {
	// 1047. 删除相邻重复项
	const stack = [];
	for (const c of s) {
		stack[stack.length - 1] === c ? stack.pop() : stack.push(c);
	}
	return stack.join('');
}
```

### 单调栈（medium 常见）

```js
function dailyTemperatures(T) {
	// 739. 每日温度
	const res = new Array(T.length).fill(0);
	const stack = []; // 存下标，栈内温度单调递减
	for (let i = 0; i < T.length; i++) {
		while (stack.length && T[i] > T[stack[stack.length - 1]]) {
			const j = stack.pop();
			res[j] = i - j;
		}
		stack.push(i);
	}
	return res;
}
```

**识别信号**：「下一个更大/更小的元素」→ 单调栈

## 六、队列与堆

### 队列

JavaScript 数组的 `shift()` 需要移动剩余元素，单次最坏是 O(n)。
算法题里可以用一个递增下标模拟队头，使出队保持 O(1)：

```js
function bfsValues(root) {
	if (!root) return [];
	const queue = [root];
	const values = [];
	let head = 0;

	while (head < queue.length) {
		const node = queue[head++];
		values.push(node.val);
		if (node.left) queue.push(node.left);
		if (node.right) queue.push(node.right);
	}
	return values;
}
```

### 最小堆（Top K）

**识别信号**：「最大的 k 个」「最小的 k 个」「动态取极值」。
下面用容量为 k 的最小堆找第 k 大元素：

```js
class MinHeap {
	constructor() {
		this.data = [];
	}

	get size() {
		return this.data.length;
	}

	push(value) {
		const data = this.data;
		data.push(value);
		let i = data.length - 1;
		while (i > 0) {
			const parent = (i - 1) >> 1;
			if (data[parent] <= data[i]) break;
			[data[parent], data[i]] = [data[i], data[parent]];
			i = parent;
		}
	}

	pop() {
		const data = this.data;
		if (data.length === 0) return undefined;
		const top = data[0];
		const last = data.pop();
		if (data.length === 0) return top;

		data[0] = last;
		let i = 0;
		while (true) {
			const left = i * 2 + 1;
			const right = left + 1;
			let smallest = i;
			if (left < data.length && data[left] < data[smallest]) smallest = left;
			if (right < data.length && data[right] < data[smallest]) smallest = right;
			if (smallest === i) break;
			[data[i], data[smallest]] = [data[smallest], data[i]];
			i = smallest;
		}
		return top;
	}

	peek() {
		return this.data[0];
	}
}

function findKthLargest(nums, k) {
	// 215. 数组中的第 K 个最大元素
	if (!Number.isInteger(k) || k < 1 || k > nums.length) {
		throw new RangeError('k 必须在 1 到数组长度之间');
	}

	const heap = new MinHeap();
	for (const num of nums) {
		heap.push(num);
		if (heap.size > k) heap.pop();
	}
	return heap.peek();
}
```

时间 O(n log k)，额外空间 O(k)。代表题：215 第 k 大、347 前 k 个高频元素、
23 合并 k 个有序链表。

## 七、链表

```js
function reverseList(head) {
	// 206. 反转链表（必会）
	let prev = null,
		cur = head;
	while (cur) {
		const next = cur.next; // 1. 先存下一个
		cur.next = prev; // 2. 反转指针
		prev = cur; // 3. prev 前进
		cur = next; // 4. cur 前进
	}
	return prev;
}

function hasCycle(head) {
	// 141. 环形链表
	let slow = head,
		fast = head;
	while (fast && fast.next) {
		slow = slow.next;
		fast = fast.next.next;
		if (slow === fast) return true;
	}
	return false;
}

function mergeTwoLists(l1, l2) {
	// 21. 合并有序链表
	const dummy = { next: null }; // 哨兵节点简化边界
	let cur = dummy;
	while (l1 && l2) {
		if (l1.val <= l2.val) {
			cur.next = l1;
			l1 = l1.next;
		} else {
			cur.next = l2;
			l2 = l2.next;
		}
		cur = cur.next;
	}
	cur.next = l1 || l2;
	return dummy.next;
}
```

**两个技巧**：

- **哨兵（dummy）节点**——避免处理头节点的特殊情况
- **快慢指针**——找环、找中点、找倒数第 k 个

## 八、二叉树

**递归三板斧**：

```js
// 前序：根→左→右  中序：左→根→右（BST 中序是升序）  后序：左→右→根
function preorder(root, res = []) {
	if (!root) return res;
	res.push(root.val);
	preorder(root.left, res);
	preorder(root.right, res);
	return res;
}

function maxDepth(root) {
	// 104. 最大深度
	return root ? 1 + Math.max(maxDepth(root.left), maxDepth(root.right)) : 0;
}

function invertTree(root) {
	// 226. 翻转二叉树
	if (!root) return null;
	[root.left, root.right] = [invertTree(root.right), invertTree(root.left)];
	return root;
}
```

### 层序遍历 BFS —— **前端最该会的一个**

```js
function levelOrder(root) {
	// 102. 层序遍历
	if (!root) return [];
	const res = [],
		queue = [root];
	let head = 0;
	while (head < queue.length) {
		const size = queue.length - head; // 进入循环时尚未访问的节点就是本层
		const level = [];
		for (let i = 0; i < size; i++) {
			const node = queue[head++]; // 避免 shift() 搬移数组元素
			level.push(node.val);
			node.left && queue.push(node.left);
			node.right && queue.push(node.right);
		}
		res.push(level);
	}
	return res;
}
```

> **为什么前端要会 BFS**：遍历 DOM 树、菜单树、多级分类——
> **业务里的树结构比 LeetCode 里多得多**。
> 见 [算法-前端专属.md](算法-前端专属.md) 的树形结构转换。

## 九、图的 BFS / DFS

**识别信号**：「连通关系」「最短步数（边权相同）」「依赖关系」「网格扩散」。
图可能有环，遍历时必须用 `visited` 防止重复访问。

```js
function bfsGraph(graph, start) {
	const queue = [start];
	const visited = new Set([start]);
	const order = [];
	let head = 0;

	while (head < queue.length) {
		const node = queue[head++];
		order.push(node);
		for (const next of graph.get(node) || []) {
			if (visited.has(next)) continue;
			visited.add(next); // 入队时标记，避免同一节点被重复入队
			queue.push(next);
		}
	}
	return order;
}

function dfsGraph(graph, start, visited = new Set(), order = []) {
	if (visited.has(start)) return order;
	visited.add(start);
	order.push(start);
	for (const next of graph.get(start) || []) {
		dfsGraph(graph, next, visited, order);
	}
	return order;
}
```

邻接表表示下，时间 O(V + E)，额外空间 O(V)。BFS 可求无权图最短路径；
带权最短路不能直接套这个模板。

## 十、排序

```js
function quickSort(nums, left = 0, right = nums.length - 1) {
	if (left >= right) return nums;

	const pivot = nums[right];
	let boundary = left;
	for (let i = left; i < right; i++) {
		if (nums[i] < pivot) {
			[nums[i], nums[boundary]] = [nums[boundary], nums[i]];
			boundary++;
		}
	}
	[nums[boundary], nums[right]] = [nums[right], nums[boundary]];

	quickSort(nums, left, boundary - 1);
	quickSort(nums, boundary + 1, right);
	return nums; // 原地修改并返回同一个数组
}

function mergeSort(arr) {
	// 归并
	if (arr.length <= 1) return arr;
	const mid = arr.length >> 1;
	const left = mergeSort(arr.slice(0, mid));
	const right = mergeSort(arr.slice(mid));
	const res = [];
	let i = 0,
		j = 0;
	while (i < left.length && j < right.length) {
		res.push(left[i] <= right[j] ? left[i++] : right[j++]);
	}
	return [...res, ...left.slice(i), ...right.slice(j)];
}
```

| 排序      | 平均       | 最坏       | 空间                     | 稳定 |
| --------- | ---------- | ---------- | ------------------------ | ---- |
| 快排      | O(n log n) | **O(n²)**  | 平均 O(log n)，最坏 O(n) | ❌   |
| 归并      | O(n log n) | O(n log n) | **O(n)**                 | ✅   |
| 堆排      | O(n log n) | O(n log n) | O(1)                     | ❌   |
| 冒泡/插入 | O(n²)      | O(n²)      | O(1)                     | ✅   |

**衍生问**：`Array.prototype.sort` 用的什么？
→ ECMAScript 从 ES2019 起要求排序结果**稳定**，但规范没有规定具体算法。
V8 当前采用 TimSort 属于引擎实现细节，不应当作所有 JavaScript 引擎的保证。

## 十一、二分查找

```js
function search(nums, target) {
	// 704. 二分查找
	let left = 0,
		right = nums.length - 1;
	while (left <= right) {
		// 闭区间 [left, right]
		const mid = left + ((right - left) >> 1);
		if (nums[mid] === target) return mid;
		if (nums[mid] < target) left = mid + 1;
		else right = mid - 1;
	}
	return -1;
}
```

**前提是搜索区间有序**。边界写法必须成套：这里使用闭区间，
所以循环条件是 `left <= right`，收缩时排除已经检查过的 `mid`。

## 十二、回溯

**识别信号**：「列出所有组合/排列/路径」。核心是：做选择 → 递归 → 撤销选择。

```js
function subsets(nums) {
	// 78. 子集
	const result = [];
	const path = [];

	function backtrack(start) {
		result.push([...path]);
		for (let i = start; i < nums.length; i++) {
			path.push(nums[i]);
			backtrack(i + 1);
			path.pop();
		}
	}

	backtrack(0);
	return result;
}
```

共有 2ⁿ 个子集；计入复制答案的成本，时间和结果空间都是 O(n · 2ⁿ)。
代表题：46 全排列、78 子集、39 组合总和、22 括号生成。

## 十三、贪心

**识别信号**：每一步选择当前最优，并且这个局部选择能够推出全局最优。
不能只凭“看起来合理”就使用贪心，需要说明或证明选择依据。

```js
function maxProfit(prices) {
	// 121. 买卖股票最佳时机
	let minPrice = Infinity;
	let profit = 0;
	for (const price of prices) {
		minPrice = Math.min(minPrice, price);
		profit = Math.max(profit, price - minPrice);
	}
	return profit;
}

function mergeIntervals(intervals) {
	// 56. 合并区间
	if (intervals.length === 0) return [];
	const sorted = intervals.map((item) => [...item]).sort((a, b) => a[0] - b[0]);
	const result = [sorted[0]];

	for (let i = 1; i < sorted.length; i++) {
		const last = result[result.length - 1];
		const current = sorted[i];
		if (current[0] <= last[1]) last[1] = Math.max(last[1], current[1]);
		else result.push(current);
	}
	return result;
}
```

`maxProfit` 是 O(n)；合并区间的主要成本是排序，因此是 O(n log n)。

## 十四、动态规划

下面三题覆盖一维状态转移和空间压缩的基础写法：

```js
function climbStairs(n) {
	// 70. 爬楼梯
	let prev = 1,
		cur = 1;
	for (let i = 2; i <= n; i++) [prev, cur] = [cur, prev + cur];
	return cur;
}

function maxSubArray(nums) {
	// 53. 最大子数组和
	if (nums.length === 0) return 0; // 这里约定空数组的结果为 0
	let max = nums[0],
		cur = nums[0];
	for (let i = 1; i < nums.length; i++) {
		cur = Math.max(nums[i], cur + nums[i]); // 要么接上，要么重开
		max = Math.max(max, cur);
	}
	return max;
}

function rob(nums) {
	// 198. 打家劫舍
	let skip = 0,
		take = 0;
	for (const num of nums) {
		[skip, take] = [Math.max(skip, take), skip + num];
	}
	return Math.max(skip, take);
}
```

**DP 思路**：状态定义 → 转移方程 → 初始值 → 能否压缩空间
