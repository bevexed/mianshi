## 一、LRU 缓存（最高频）

**为什么前端要会**：带容量上限的组件缓存、图片缓存和接口缓存，
都可能用 LRU 作为淘汰策略。

**JS 的取巧解法——利用 `Map` 有序**：

```js
class LRUCache {
	constructor(capacity) {
		if (!Number.isInteger(capacity) || capacity < 1) {
			throw new RangeError('capacity 必须是正整数');
		}
		this.capacity = capacity;
		this.map = new Map(); // ECMAScript 规定 Map 按插入顺序迭代
	}

	get(key) {
		if (!this.map.has(key)) return -1;
		const val = this.map.get(key);
		this.map.delete(key); // 删了再插 = 移到最新
		this.map.set(key, val);
		return val;
	}

	put(key, value) {
		if (this.map.has(key)) this.map.delete(key);
		this.map.set(key, value);
		if (this.map.size > this.capacity) {
			this.map.delete(this.map.keys().next().value); // 删最旧的
		}
	}
}
```

**⚠️ 面试官可能追问**：不用 Map，用双向链表 + 哈希表怎么写？
→ 说明思路即可：**哈希表 O(1) 定位节点，双向链表 O(1) 移动和删除**。
这里利用的是 `Map` 的公开语义，不需要依赖某个 JavaScript 引擎的内部结构。

## 二、数组扁平化

```js
// 1. 内置（能用就用）
arr.flat(Infinity);

// 2. 递归
function flatten(arr) {
	return arr.reduce((acc, cur) => acc.concat(Array.isArray(cur) ? flatten(cur) : cur), []);
}

// 3. 迭代（避免栈溢出，数据极深时用）
function flattenIter(arr) {
	const stack = [...arr].reverse();
	const res = [];
	while (stack.length) {
		const item = stack.pop();
		if (Array.isArray(item)) {
			// 逆序压栈，弹出时才能保持原数组的从左到右顺序
			for (let i = item.length - 1; i >= 0; i--) stack.push(item[i]);
		} else {
			res.push(item);
		}
	}
	return res;
}

// 4. 指定深度
function flattenDepth(arr, depth = 1) {
	return depth > 0
		? arr.reduce(
				(acc, cur) => acc.concat(Array.isArray(cur) ? flattenDepth(cur, depth - 1) : cur),
				[]
		  )
		: arr.slice();
}
```

## 三、树形结构转换（业务里最常用）

### 扁平数组 → 树

```js
function listToTree(list, rootId = null) {
	const map = new Map();
	const roots = [];

	// 第一遍：建索引
	for (const item of list) {
		map.set(item.id, { ...item, children: [] });
	}
	// 第二遍：挂载
	for (const item of list) {
		const node = map.get(item.id);
		if (item.parentId === rootId || !map.has(item.parentId)) {
			roots.push(node);
		} else {
			map.get(item.parentId).children.push(node);
		}
	}
	return roots;
}
```

**关键点**：**两次遍历，O(n)**。
一次遍历的写法要求数据有序（父在子前），不可靠。

**业务场景**：菜单树、部门树、多级分类、评论嵌套；
ERP 和后台管理系统里尤其常见。

### 树 → 扁平数组

```js
function treeToList(tree) {
	const res = [];
	const stack = [...tree].reverse();
	while (stack.length) {
		const { children, ...rest } = stack.pop();
		res.push(rest);
		if (children) {
			// 栈是后进先出，逆序压入才能得到稳定的先序遍历
			for (let i = children.length - 1; i >= 0; i--) stack.push(children[i]);
		}
	}
	return res;
}
```

这组实现约定 `id` 唯一、输入关系中没有环；找不到父节点的项作为根节点返回。
如果业务要求孤儿节点报错或需要检测环，应当在建索引后单独校验，而不是静默挂载。

### 树中查找路径（面包屑）

```js
function findPath(tree, targetId, path = []) {
	for (const node of tree) {
		const newPath = [...path, node];
		if (node.id === targetId) return newPath;
		if (node.children) {
			const found = findPath(node.children, targetId, newPath);
			if (found) return found;
		}
	}
	return null;
}
```

## 四、大数相加

**为什么会考**：超过 `Number.MAX_SAFE_INTEGER` 后，JavaScript `Number`
不能保证每个整数都被精确表示——**订单号、雪花 ID 不应当按 Number 接收**。

```js
function addBigNumber(a, b) {
	if (!/^\d+$/.test(a) || !/^\d+$/.test(b)) {
		throw new TypeError('只支持非负整数字符串');
	}
	let i = a.length - 1,
		j = b.length - 1;
	let carry = 0;
	const res = [];
	while (i >= 0 || j >= 0 || carry) {
		const x = i >= 0 ? Number(a[i--]) : 0;
		const y = j >= 0 ? Number(b[j--]) : 0;
		const sum = x + y + carry;
		res.push(sum % 10);
		carry = Math.floor(sum / 10);
	}
	const value = res
		.reverse()
		.join('')
		.replace(/^0+(?=\d)/, '');
	return value;
}

addBigNumber('9007199254740993', '1'); // '9007199254740994'
```

**衍生的实际问题**：

> 后端返回的 Long 型 ID（如雪花 ID）在 JS 里可能丢失精度。
> **解法是后端序列化时转成字符串**，前端用 string 接。
>
> 这是常见的联调坑，能同时说明原因和接口约定会更有说服力。

见 [16-后端常识/后端分层与接口设计.md](../../16-后端常识/后端分层与接口设计.md)

## 五、千分位格式化

```js
// 1. 正则（面试要能写出来）
function formatMoney(n) {
	return String(n).replace(/\B(?=(\d{3})+(?!\d))/g, ',');
}

// 2. 保留原小数文本的版本（输入需是普通十进制文本，不支持科学计数法）
function formatMoney2(n) {
	const [int, dec] = String(n).split('.');
	const formatted = int.replace(/\B(?=(\d{3})+(?!\d))/g, ',');
	return dec ? `${formatted}.${dec}` : formatted;
}

// 3. 实际项目用内置
new Intl.NumberFormat('zh-CN').format(1234567.89); // '1,234,567.89'
```

**正则解释**（会被追问）：

- `\B` —— 非单词边界（防止在开头加逗号）
- `(?=...)` —— 正向先行断言，只匹配位置不消费字符
- `(\d{3})+` —— 后面是 3 的倍数个数字
- `(?!\d)` —— 且再后面不是数字

**实际项目建议用 `Intl.NumberFormat`**——
它还能处理不同地区的格式（德国是 `1.234.567,89`）。

> 金额、时区和接口传输约定见 [表设计与常见坑](../../15-数据库/表设计与常见坑.md)。

## 六、版本号比较

```js
function compareVersion(v1, v2) {
	const a = v1.split('.'),
		b = v2.split('.');
	const len = Math.max(a.length, b.length);
	for (let i = 0; i < len; i++) {
		const raw1 = a[i] ?? '0';
		const raw2 = b[i] ?? '0';
		if (!/^\d+$/.test(raw1) || !/^\d+$/.test(raw2)) {
			throw new TypeError('只支持点分隔的非负整数版本号');
		}
		const n1 = raw1.replace(/^0+(?=\d)/, '');
		const n2 = raw2.replace(/^0+(?=\d)/, '');
		if (n1.length !== n2.length) return n1.length > n2.length ? 1 : -1;
		if (n1 !== n2) return n1 > n2 ? 1 : -1;
	}
	return 0;
}

compareVersion('1.2.10', '1.2.9'); // 1  ← 注意不能按字符串比
```

**考点**：不能直接比较整个字符串（`'1.2.10' < '1.2.9'` 是 true，错的），
且要处理位数不同（`1.2` vs `1.2.0`）和超长数字段。
这个版本只处理数字版本号；完整 SemVer 还需要处理预发布标识和构建元数据。

## 七、URL 参数解析

```js
// 手写版
function parseQuery(url) {
	const query = Object.create(null);
	const questionIndex = url.indexOf('?');
	const search = questionIndex === -1 ? '' : url.slice(questionIndex + 1).split('#')[0];
	if (!search) return query;
	for (const pair of search.split('&')) {
		if (!pair) continue;
		const equalIndex = pair.indexOf('=');
		const rawKey = equalIndex === -1 ? pair : pair.slice(0, equalIndex);
		const rawValue = equalIndex === -1 ? '' : pair.slice(equalIndex + 1);
		const decode = (value) => decodeURIComponent(value.replace(/\+/g, ' '));
		const key = decode(rawKey);
		const val = decode(rawValue);
		// 处理同名参数：a=1&a=2 → {a: ['1','2']}
		if (Object.prototype.hasOwnProperty.call(query, key)) {
			query[key] = [].concat(query[key], val);
		} else {
			query[key] = val;
		}
	}
	return query;
}

// 实际项目优先保留 URLSearchParams；直接 fromEntries 会丢失同名参数的前几项
const searchParams = new URLSearchParams(location.search);
searchParams.getAll('a');
```

**考点**：解码、`+`、同名参数、空值、值中的 `=`、hash 干扰。
示例使用无原型对象，避免 `__proto__` 等特殊键触发原型上的 setter；
非法百分号编码会按 `decodeURIComponent` 的语义抛出 `URIError`。

## 九、并发控制池（高频）

```js
async function pool(tasks, limit = 4) {
	if (!Number.isInteger(limit) || limit < 1) {
		throw new RangeError('limit 必须是正整数');
	}

	const results = new Array(tasks.length);
	let nextIndex = 0;

	async function worker() {
		while (true) {
			const index = nextIndex++;
			if (index >= tasks.length) return;
			// 放进 then 中调用，任务同步抛错也会变成 Promise 拒绝
			results[index] = await Promise.resolve().then(tasks[index]);
		}
	}

	const workerCount = Math.min(limit, tasks.length);
	await Promise.all(Array.from({ length: workerCount }, () => worker()));
	return results;
}
```

**业务场景**：大文件分片上传、批量请求限流。
结果按输入顺序保存；任一任务失败时返回的 Promise 会拒绝，已经启动的任务无法自动取消。
需要取消时应让任务接收同一个 `AbortSignal`，并在失败分支中触发 `abort()`。

> 见 [12-场景题/大文件上传与断点续传.md](../../12-场景题/大文件上传与断点续传.md)

## 十、对象扁平化

```js
function flattenObject(input) {
	if (input === null || typeof input !== 'object' || Array.isArray(input)) {
		throw new TypeError('input 必须是对象');
	}

	const result = Object.create(null);

	function visit(value, path) {
		if (value === null || typeof value !== 'object') {
			result[path] = value;
			return;
		}

		const keys = Object.keys(value);
		if (keys.length === 0) {
			result[path] = Array.isArray(value) ? [] : {};
			return;
		}

		for (const key of keys) {
			visit(value[key], path ? `${path}.${key}` : key);
		}
	}

	for (const key of Object.keys(input)) visit(input[key], key);
	return result;
}

flattenObject({ a: { b: 1 }, list: ['x', 'y'] });
// { 'a.b': 1, 'list.0': 'x', 'list.1': 'y' }
```

这里约定输入是没有循环引用的 JSON 风格对象；数组下标也会进入路径。
如果键本身可能包含 `.`，需要改用路径数组或定义转义规则，不能直接拼字符串。

## 十一、柯里化与函数组合

```js
function curry(fn, arity = fn.length) {
	if (!Number.isInteger(arity) || arity < 0) {
		throw new RangeError('arity 必须是非负整数');
	}
	function curried(...args) {
		if (args.length >= arity) return fn.apply(this, args);
		return function collect(...nextArgs) {
			return curried.apply(this, [...args, ...nextArgs]);
		};
	}
	return curried;
}

const pipe =
	(...fns) =>
	(input) =>
		fns.reduce((value, fn) => fn(value), input);
const compose =
	(...fns) =>
	(input) =>
		fns.reduceRight((value, fn) => fn(value), input);

const add = (a, b, c) => a + b + c;
curry(add)(1)(2, 3); // 6
pipe(
	(x) => x + 1,
	(x) => x * 2
)(3); // 8
compose(
	(x) => x + 1,
	(x) => x * 2
)(3); // 7
```

`curry` 默认用 `fn.length` 判断参数是否收集完成；带默认参数或剩余参数的函数
需要通过第二个参数显式传入目标参数个数。这里的 `pipe` / `compose` 是一元同步版本；
异步流水线需要在 `reduce` 中串联 Promise。

## 面试策略

**这类题的正确打开方式**：

1. **先问清边界**——「输入一定是合法的吗」「要考虑循环引用吗」
   （问边界本身就是加分项）
2. **先写能跑的版本**，再谈优化
3. **主动说出复杂度和局限**——
   「这个版本没处理 X，如果需要我可以改」
4. **能联系到业务最好**——
   「这个我在做树形菜单的时候实际用过」

如果确实在项目里使用过，第 4 点会比只背模板更有说服力；
没有用过就说明自己理解的适用场景，不要把练习题包装成项目经历。
